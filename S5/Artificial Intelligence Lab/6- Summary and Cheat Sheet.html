The Decision Tree — Which Algorithm to Use?

Is it a TWO-PLAYER GAME (Tic-Tac-Toe, Chess)?
→ Minimax (or Alpha-Beta Pruning for speed)
↓ No, it's a path/search problem
Do you have a HEURISTIC (estimate to goal)?
→ Yes: Use A* (optimal) or Greedy (fast but not optimal)
→ No: Use blind search below
↓ No heuristic, blind search
Does the graph have EDGE COSTS (weighted)?
→ Yes: Use UCS (cheapest path)
→ No: Continue below
↓ Unweighted graph
Do you need the SHORTEST path?
→ Yes + low memory: IDS | Yes + memory OK: BFS
→ No (any path): DFS
↓ Optimization (find best value, not path)
Large state space, need best value (not path)?
→ Hill Climbing (+ Random Restart if needed)

Key Formula Summary

UCS: g(n)
Actual cost from start
Greedy: h(n)
Estimated cost to goal
A*: g(n)+h(n)
Actual + estimated

3Complete Code Templates

Copy-paste-ready templates for every algorithm — study the structure

Template — Node Class (use for ALL search algorithms)
class Node:
    def __init__(self, state, parent=None, cost=0, depth=0):
        self.state = state
        self.parent = parent
        self.cost = cost
        self.depth = depth
    def __lt__(self, other):      # needed for heapq
        return self.cost < other.cost

def get_path(node):               # reconstruct path from goal to start
    path = []
    while node:
        path.append(node.state)
        node = node.parent
    return path[::-1]             # reverse: start→goal
Template — BFS (Queue/FIFO)
from collections import deque

def bfs(graph, start, goal):
    queue = deque([Node(start)])
    visited = set()
    while queue:
        n = queue.popleft()              # ← FRONT (FIFO)
        if n.state == goal: return get_path(n)
        if n.state in visited: continue
        visited.add(n.state)
        for nb in graph[n.state]:
            if nb not in visited:
                queue.append(Node(nb, n, depth=n.depth+1))
    return None
Template — DFS (Stack/LIFO)
def dfs(graph, start, goal):
    stack = [Node(start)]
    visited = set()
    while stack:
        n = stack.pop()                  # ← TOP (LIFO)
        if n.state == goal: return get_path(n)
        if n.state in visited: continue
        visited.add(n.state)
        for nb in graph[n.state]:
            if nb not in visited:
                stack.append(Node(nb, n, depth=n.depth+1))
    return None
Template — UCS (Priority Queue / min-heap)
import heapq

def ucs(graph, start, goal):
    pq = []; heapq.heappush(pq, (0, Node(start, cost=0)))
    visited = {}
    while pq:
        cost, n = heapq.heappop(pq)      # ← LOWEST COST first
        if n.state == goal:
            print(f"Cost: {cost}"); return get_path(n)
        if n.state in visited and visited[n.state] <= cost: continue
        visited[n.state] = cost
        for nb, ec in graph[n.state]:
            nc = cost + ec
            heapq.heappush(pq, (nc, Node(nb, n, cost=nc)))
    return None
Template — IDS (Iterative Deepening)
def depth_limited_dfs(node, graph, goal, limit):
    if node.state == goal: return node
    if node.depth >= limit: return None
    for nb in graph[node.state]:
        child = Node(nb, node, depth=node.depth+1)
        result = depth_limited_dfs(child, graph, goal, limit)
        if result: return result
    return None

def ids(graph, start, goal, max_depth=20):
    for depth in range(max_depth):
        result = depth_limited_dfs(Node(start), graph, goal, depth)
        if result: return get_path(result)
    return None
Template — A* Search
import heapq

def astar(graph, heuristic, start, goal):
    pq = []; heapq.heappush(pq, (heuristic[start], Node(start, cost=0)))
    visited = {}
    while pq:
        f, n = heapq.heappop(pq)
        g = n.cost
        if n.state == goal:
            print(f"Cost: {g}"); return get_path(n)
        if n.state in visited and visited[n.state] <= g: continue
        visited[n.state] = g
        for nb, ec in graph[n.state]:
            ng = g + ec
            nf = ng + heuristic[nb]           # f = g + h
            heapq.heappush(pq, (nf, Node(nb, n, cost=ng)))
    return None
Template — Minimax with Alpha-Beta Pruning
class GameNode:
    def __init__(self, value=None, children=None):
        self.value = value
        self.children = children or []

def minimax(node, is_max):
    if not node.children: return node.value
    if is_max:
        return max(minimax(c, False) for c in node.children)
    else:
        return min(minimax(c, True)  for c in node.children)

def alpha_beta(node, is_max, alpha=float('-inf'), beta=float('inf')):
    if not node.children: return node.value
    if is_max:
        best = float('-inf')
        for c in node.children:
            best = max(best, alpha_beta(c, False, alpha, beta))
            alpha = max(alpha, best)
            if beta <= alpha: break          # Beta cut-off
        return best
    else:
        best = float('inf')
        for c in node.children:
            best = min(best, alpha_beta(c, True, alpha, beta))
            beta = min(beta, best)
            if beta <= alpha: break          # Alpha cut-off
        return best
Template — Hill Climbing
def hill_climbing(landscape, start):
    current = start
    while True:
        neighbours = landscape.get(current, [])
        if not neighbours: break
        best = max(neighbours)        # steepest-ascent variant
        if best > current:
            current = best
        else:
            break                     # local maximum — stop
    return current

def random_restart_hc(landscape, num_restarts=5):
    import random
    states = list(landscape.keys())
    best = None
    for _ in range(num_restarts):
        start = random.choice(states)
        result = hill_climbing(landscape, start)
        if best is None or result > best:
            best = result
    return best

def hc_function(f, start_x, step=1):
    x = start_x
    while True:
        neighbours = [x - step, x + step]
        best_x = max(neighbours, key=f)
        if f(best_x) > f(x): x = best_x
        else: break
    return x, f(x)
Template — Genetic Algorithm (GA) with Ranking & Mutation
import random

def ga_optimize(fitness_func, pop_size=10, gens=10, mutation_rate=0.2):
    pop = [random.uniform(-5, 15) for _ in range(pop_size)]
    for _ in range(gens):
        # 1. Evaluate & Rank (highest fitness first)
        ranked = sorted(pop, key=fitness_func, reverse=True)
        # 2. Select top 50% (ranking selection)
        survivors = ranked[:pop_size // 2]
        # 3. Duplicate survivors to refill population
        pop = (survivors * 2)[:pop_size]
        # 4. Controlled mutation: x = x + uniform(-1, 1) with prob 0.2
        for i in range(len(pop)):
            if random.random() < mutation_rate:
                pop[i] += random.uniform(-1, 1)
    best = max(pop, key=fitness_func)
    return best, fitness_func(best)

4Exam Day Checklist

Everything to remember when you sit down to code

Before You Write Any Code

Common Exam Mistakes to Avoid

Avoid These
  • Using list.pop(0) for BFS — use deque.popleft() instead
  • Forgetting __lt__ in Node when using heapq (UCS, Greedy, A*)
  • Not reversing the path in get_path — always do path[::-1]
  • Not adding goal check before the visited check (UCS/A*)
  • Using {} for empty set — use set()
  • Forgetting t = t.sort() returns None — use sorted(t)
  • Accessing dict key directly without checking: d[k] — use d.get(k)
  • Forgetting trailing comma in single-item tuple: (x,) not (x)

Algorithm Skeletons — Fill-In-the-Blank

BFS vs DFS — Only 1 Line Different
# BFS  →  deque  +  popleft()  →  FIFO  →  level-by-level
# DFS  →  list   +  pop()      →  LIFO  →  depth-first

# UCS  →  heapq  +  (cost, node)  →  cheapest first
# A*   →  heapq  +  (g+h, node)   →  g(n)+h(n) first

OOP in 20 Lines
class Animal:
    species = "Unknown"           # class variable (shared)
    def __init__(self, name):
        self.name = name          # instance variable (unique)
    def speak(self): print(f"{self.name} speaks")
    @classmethod
    def set_species(cls, s): cls.species = s
    @staticmethod
    def is_valid(name): return len(name) > 0

class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name)    # MUST call parent init
        self.breed = breed
    def speak(self):              # method override
        print(f"{self.name} says WOOF")
    def fetch(self): print(f"{self.name} fetches!")

class GuideDog(Dog):              # multilevel: GuideDog→Dog→Animal
    def __init__(self, name, breed, owner):
        super().__init__(name, breed)
        self.owner = owner
    def guide(self): print(f"Guiding {self.owner}")

FINAL SUMMARY — The 10 Things You Must Know

  • 1. Counting pattern: counts[x] = counts.get(x,0) + 1
  • 2. List comprehension: [f(x) for x in lst if cond]
  • 3. Sort by value: sorted(d.items(), key=lambda x:x[1], reverse=True)
  • 4. BFS = deque + popleft (FIFO). DFS = list + pop (LIFO)
  • 5. UCS: push (cost, node) to heapq, expand cheapest first
  • 6. A* = g(n) + h(n). Greedy = h(n) only. UCS = g(n) only
  • 7. All search needs: Node class + visited set + get_path()
  • 8. Minimax: MAX picks max of children, MIN picks min. Alternates.
  • 9. Alpha-Beta: prune when β ≤ α. Same result as Minimax.
  • 10. Hill Climbing: move to best neighbour, stop at local max. Fix: Random Restart.
